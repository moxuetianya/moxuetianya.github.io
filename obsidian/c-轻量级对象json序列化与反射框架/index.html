一、项目背景

C++没有原生的反射与序列化，至少从语言层面来说是暂时不支持的 很多时候是需要自己手动去写反射相关的代码 自己实现方案很多，也有很多现成的库可以使用 但是很多代码用起来都异常繁琐，包含各种模板以及宏，比如类似这样的

总之我觉得无论是使用也好，还是注册也好，用起来都是很繁琐的。 所以我自然希望实现一个类似于Tensorflow的自动微分那样，只要写好前向传播代码，自动微分就已经自动生成好了。 我希望当我写完CPP的代码，对应的struct/class的序列化、反序列化、反射就已经自动完成了所有注册， 也不用单独为每个class写一个序列化与反序列化，一切东西都用起来足够"原生"。 所以我的代码尽可能简化了代码使用的这一过程，具体实现其实也不复杂，就是各种类型匹配和递归调用的问题。 我将会在后文中给出充分的示例。

总之就是我希望能有一个更简洁的方案，能够[自动]完成反射、序列化、反序列化。就像深度学习里面的自动微分框架一样，只要我构建好代码，就已经自动构建好了微分运算。只要我写好了结构体，就自动构建好了对应的序列化、反序列化方法。

使用起来就像这样，非常的简洁，简洁得像是在写python

当然可序列化的前提就是可反射的，所以Serializable是Reflectable的子类

二、相关限制

原本反射只是为了序列化和反序列化，只能支持属性值，现在已经支持成员函数了，不过并未处理继承与派生，以及函数重载的问题，所以功能上还比较弱鸡。序列化上可以支持单继承

目前我的代码支持属性满足以下类型条件的class的序列化与反序列化

(1).C++基本类型(int,float,char,...)

(2).含有迭代器的容器(std::vector,std::list,std::deque,...)

(3).容器适配器(std::map,std::set,std::unordered_map,std::unordered_set,...)

(4).std::tuple和std::pair

(5).其他正确实现了get_config以及调用Serializable::regist<T>注册的类

(6).数组(int a[15]),std::array<int,15>

(7).上述类型的组合及其指针(std::vector[std::pair**int\*,float\*](int%5C*,float%5C*%5D(std::pair%3Cint%5C*,float%5C*))** )，可以正确处理空指针，空指针将被序列化为"null"。

(8).派生类的序列化与反序列化(暂没考虑多继承虚继承的问题)

四、序列化、反序列化代码示例

序列化，反序列化关键方法只有两个 Serializable::dumps和Serializable::loads 分别对应序列化与反序列化，其中反序列化需要显式的给出被反序列化的对象类型 如果不给的话将会返回一个void*指针，如果给了参数，会返回一个引用 如果不想调用额外的复制构造函数，可以使用auto&object=Serializable::loads(json_string)

(1).trie字典树的序列化与反序列化

首先我们实现一棵朴素的trie树，没经过任何优化，这里面包括了std::vector,Node*,bool多种类型需要处理。

#include<iostream>
#include<fstream>
#include<vector>
using namespace std;
struct Node
{
	Node():sons(26,nullptr),flag(false){};
	~Node();
	std::vector<Node*>sons;
	bool flag;
};
class Trie
{
public:
	Trie():root(new Node){}
	void insert(const std::string&str);
	void print();
private:
	void __print(Node*root,std::string&str);
	Node*root;
};
Node::~Node()
{
	for(auto&son:sons)
		if(son!=nullptr)
			delete son;
}
void Trie::print()
{
	std::string tmp;
	__print(root,tmp);
}
void Trie::insert(const std::string&str)
{
	Node*root=this->root;
	for(auto&it:str)
	{
		if(root->sons[it-'a']==nullptr)
			root->sons[it-'a']=new Node();
		root=root->sons[it-'a'];
	}
	root->flag=true;
}
void Trie::__print(Node*root,std::string&str)
{
	if(root->flag>0)
	{
		cout<<str<<endl;
	}
	for(unsigned i=0;i<root->sons.size();i++)
	{
		if(root->sons[i]!=nullptr)
		{
			str.push_back(i+'a');
			__print(root->sons[i],str);
			str.pop_back();
		}
	}
}
int main()
{
	ifstream fin("test.json",ios::in);
	ofstream fout("test.json",ios::out);
	Trie trie;
	trie.insert("ababa");
	trie.insert("abab");
	trie.insert("xisa");
	trie.insert("cpp");                                        //构建一棵字典树
	trie.insert("cest");
	trie.insert("test");
	trie.insert("tet");
	trie.print();                                              //打印trie树所有单词
}

struct Node:public Serializable
{
	Node(Node*lson,Node*rson,std::function<int(Node*,Node*)>eval):value(eval(lson,rson)),lson(lson),rson(rson){}
	Node(int value=0):value(value),lson(nullptr),rson(nullptr){}
	~Node()
	{
		if(lson!=nullptr)
			delete lson;
		if(rson!=nullptr)
			delete rson;
	}
	Config get_config()const //变化1
	{
		Config config=Serializable::get_config(this);
		config.update({
			{"value",value},
			{"lson",lson},
			{"rson",rson}
		});
		return config;
	}
	int value;
	Node*lson;
	Node*rson;
};
int main()
{
	Serializable::Regist<Node>(); //变化2
	ifstream fin("test.json",ios::in);
	ofstream fout("test.json",ios::out);
	auto add=Node*x,Node*y{return x->value+y->value;};
	auto sub=Node*x,Node*y{return x->value-y->value;};
	auto mul=Node*x,Node*y{return x->value*y->value;};
	Node*x=new Node(5);
	Node*y=new Node(6);
	Node*z=new Node(x,y,add); //z=x+y=11
	z=new Node(x,z,mul);      //z=x*z=55
	z=new Node(z,y,sub);      //z=z-y=49
	std::string json=Serializable::dumps(*z);  //序列化
	fout<<json;
	fout.close();

	getline(fin,json);
	fin.close();
	cout<<"json string:\n"<<json<<endl<<endl;
	Node*root=(Node*)Serializable::loads(json); //反序列化
	cout<<root->value;
}

(7).派生类的序列化与反序列化

#include"lang/serializable.h"
#include"iostream"
using namespace std;
struct Father
{
	int x=1;
	std::vector<int>y{1,2,3,4};
	virtual Config get_config()const
	{
		Config config=Serializable::get_config(this);
		config.update({
			{"x",x},
			{"y",y}
		});
		return config;
	}
	virtual ~Father(){}
};
struct Son:public Father
{
	float z[4]={5,6,7,8};
	Config get_config()const
	{
		Config config=Serializable::Inherit<Father>::get_config(this);
		config.update({
			{"z",z}
		});
		return config;
	}
};
struct GrandSon:public Son
{
	std::tuple<int,std::string>t={2021,"shijunfeng00"};
	Config get_config()const
	{
		Config config=Serializable::Inherit<Son>::get_config(this);
		config.update({
			{"t",t}
		});
		return config;
	}
};
int main()
{
	Serializable::Regist<Father,Son,GrandSon>();
	Father*son=new GrandSon();
	son->x=233;
	std::string json=Serializable::dumps(*son);
	cout<<"json string:\n"<<json<<endl;
	GrandSon*gs=(GrandSon*)Serializable::loads(json);
	cout<<gs->get_config().serialized_to_string(true)<<endl;
	cout<<gs->x<<endl;
	for(auto&it:gs->y)
		cout<<it<<",";
	cout<<endl;
	for(auto&it:gs->z)
		cout<<it<<",";
	cout<<endl;
	cout<<get<0>(gs->t)<<" "<<get<1>(gs->t);
}

六、反射的代码示例

反射相对于序列化/反序列化会多一些细节出来,如果只需要反射,可以只用#include"lang/reflectable.h"

以及对应的Reflectable::Regist()和Config config=Reflectable::get_config(this)

这样的话就只有反射没有序列化/反序列化的功能

代码示例(1)：打印属性名称和方法名称

#include"lang/reflectable.h"
#include<iostream>
#include<vector>
using namespace std;
struct Node
{
    int x=1;
    float y=5;
    std::string z="sjf";
    int add(int x,int y)
    {
        return x+y;
    }
    std::string getName()
    {
        return z;
    }
    Config get_config()const
    {
        Config config=Reflectable::get_config(this);
        config.update({
            {"x",x},
            {"y",y},
            {"z",z},
            {"add",add},
            {"getName",getName}
        });
        return config;
    }
};
int main()
{
    Reflectable::Regist<Node>();
    Node a;
    cout<<"fields:\n";
    for(auto&it:Reflectable::get_field_names<Node>())
        cout<<"name:"<<it<<"    type:"<<Reflectable::get_field_type("Node",it)<<endl;  //打印属性名称
    cout<<"methods:\n";
    for(auto&it:Reflectable::get_method_names<Node>())                                     //打印方法名称
        cout<<"name:"<<it<<endl;
}

代码中可以使用Node*object=*(Node*)Reflectable::get_instance("Node")来得到一个Node对象，使用Reflectable::get_field来获得属性，对于private属性也能正常访问。

  • 对于有类型的对象，比如上文的Node a，可以采用int&field=Reflectable::get_field<int>(a,"x");来得到属性的引用， 也可以使用void*field=Reflectable::get_field(a,"x");来得到一个void*指针。
  • 对于void*指针表示的对象，比如void*a=Reflectable::get_instance("Node")得到的对象，需要给出具体类型的字符串名称， 即int&field2=Reflectable::get_field<int>(a,"Node","x");， 或者void*field1=Reflectable::get_field(a,"Node","x"); 同样代码里面也可以使用Reflectable::set_field来直接设置属性类型
  • 使用Reflectable::set_field(a,"x",5);或者Reflectable::set_field<int>(a,"x",5); 可以显式给出类型，也可以由输入参数自动推断。
  • 对于void*类型，依然需要给出字符串名称Reflectable::set_field<int>(a,"Node","x",5); 对于成员函数的反射，需要显式给出函数的返回值类型Reflectable::get_method<string>(b,"add",5,6); 第一个参数是对象实例，然后是函数名称，然后是函数的参数列表

#include<iostream>

#include<vector>

struct Node
{
    int x=1;
    float y=5;
    std::string z="sjf";
    int add(int x,int y)
    {
        return x+y;
    }
    std::string getName()
    {
        return z;
    }
    Config get_config()const
    {
        Config config=Reflectable::get_config(this);
        config.update({
            {"x",x},
            {"y",y},
            {"z",z},
            {"add",add},
            {"getName",getName}
        });
        return config;
    }
};
struct Point
{
    float x=0,y=0;
    Point add(float delta)
    {
        Point p(*this);
        p.x+=delta;
        p.y+=delta;
        return p;
    }
    Config get_config()const
    {
        Config config=Reflectable::get_config(this);
        config.update({
            {"x",x},
            {"y",y},
            {"add",add},
        });
        return config;
    }
};
int main()
{
    /*构建实例*/
    Reflectable::Regist<Node,Point>();
    void*a=Reflectable::get_instance("Node");                        //创建实例a
    Node b;                                                          //创建实例b
    Point c;
    /*属性值反射*/
    Reflectable::set_field(a,"Node","x",4);                          //不显式给出类型,由第三个参数来推断
    Reflectable::set_field<float>(a,"Node","y",5);
    Reflectable::set_field<string>(a,"Node","z","test");

    int field_x=Reflectable::get_field<int>(a,"Node","x");           //通过字符串名称得到值
    float field_y=Reflectable::get_field<float>(a,"Node","y");
    string field_z=Reflectable::get_field<string>(a,"Node","z");

    cout<<field_x<<" "<<(*(Node*)a).x<<endl;                         //正常访问成员变量
    cout<<field_y<<" "<<(*(Node*)a).y<<endl;
    cout<<field_z<<" "<<(*(Node*)a).z<<endl;

    Reflectable::set_field<int>(b,string("x"),10);
    Reflectable::set_field<float>(b,"y",15);

    field_x=Reflectable::get_field<int>(b,"x");
    field_y=Reflectable::get_field<float>(b,"y");

    cout<<field_z<<" "<<b.z<<endl;

    cout<<Reflectable::get_method<int>(b,"add",6,7)<<endl;
    cout<<Reflectable::get_method<string>(b,"getName")<<endl;
